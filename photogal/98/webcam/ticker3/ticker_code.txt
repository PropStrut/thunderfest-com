
Ticker code examples. These are copies of the
code used on the Java hermit site tickers demo...


HTick (horizontal scroll)...

<APPLET CODE="HTick10.class" WIDTH=240 HEIGHT=32>

  <PARAM NAME=BORDERWIDTH VALUE=5>
  <PARAM NAME=BORDERSTYLE VALUE=3>
  <PARAM NAME=BORDERCOLOR VALUE=#00CCCC>
  <PARAM NAME=TEXTFONT VALUE="TimesRoman,3,18">
  <PARAM NAME=TEXTCOLOR VALUE=#FFFFFF>
  <PARAM NAME=BACKCOLOR VALUE=#000066>
  <PARAM NAME=SCROLLSPEED VALUE=30>
  <PARAM NAME=CAPTIONSPACE VALUE=10>
  <PARAM NAME=CAPTION1 VALUE="The hermit presents...">
  <PARAM NAME=CAPTION2 VALUE="a scrolling ticker applet...">
  <PARAM NAME=CAPTION3 VALUE="better than a MARQUEE...">
  <PARAM NAME=CAPTION4 VALUE="and it's FREE!">

</APPLET>
